**Fix unsafe JSON construction for `reason` / `context` strings**
`reason` and `context` are injected into JSON with `printf`:
```bash
reason="$(printf '%s' "$hook_output" | jq -r '.reason // "Operation blocked by task guard"')"
printf '{"decision": "block", "reason": "%s"}\n' "$reason"
...
context="$(printf '%s' "$hook_output" | jq -r '.context // empty')"
printf '{"decision": "approve", "context": "%s"}\n' "$context"
```
If either value contains quotes, backslashes, or newlines, the output is not valid JSON. Since hook_output is already JSON, it’s safer to let `jq` handle encoding:
```diff
-# Check decision
-decision="$(printf '%s' "$hook_output" | jq -r '.decision // "approve"')"
-
-if [[ "$decision" == "block" ]]; then
-  # Return the block reason to Claude
-  reason="$(printf '%s' "$hook_output" | jq -r '.reason // "Operation blocked by task guard"')"
-  printf '{"decision": "block", "reason": "%s"}\n' "$reason"
-else
-  # Approve - optionally include context
-  context="$(printf '%s' "$hook_output" | jq -r '.context // empty')"
-  if [[ -n "$context" ]]; then
-    printf '{"decision": "approve", "context": "%s"}\n' "$context"
-  else
-    printf '{}\n'
-  fi
-fi
+# Check decision and render JSON via jq to ensure proper escaping
+decision="$(printf '%s' "$hook_output" | jq -r '.decision // "approve"')"
+
+if [[ "$decision" == "block" ]]; then
+  printf '%s\n' "$hook_output" | jq -c '{
+    decision: "block",
+    reason: (.reason // "Operation blocked by task guard")
+  }'
+else
+  printf '%s\n' "$hook_output" | jq -c '
+    if (.context // "") != "" then
+      { decision: "approve", context: .context }
+    else
+      {}
+    end
+  '
+fi
```
This preserves behavior while guaranteeing valid JSON.

import "math"
func cosineSimilarity(a, b []float32) float32 {
    var dot, normA, normB float64
    for i := range a {
        av, bv := float64(a[i]), float64(b[i])
        dot += av * bv
        normA += av * av
        normB += bv * bv
    }
    if normA == 0 || normB == 0 {
        return 0
    }
    return float32(dot / (math.Sqrt(normA) * math.Sqrt(normB)))
}
